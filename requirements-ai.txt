# AI edition — adds the two foundation models on top of requirements.txt.
#
# REQUIRES PYTHON >= 3.10. Both packages declare it, so on Python 3.9 pip will
# simply refuse to install them and the app silently stays in Standard mode.
#
# These pull in torch, which is roughly 800 MB unpacked on Windows and turns a
# ~120 MB portable app into a ~900 MB one. That is why they are a separate
# edition rather than the default: most pharmacy owners forecasting monthly
# revenue are well served by ARIMA and ETS, and a 1 GB download is a real
# barrier for a tool meant to be double-click simple.
#
#   pip install -r requirements.txt -r requirements-ai.txt
#
# Model weights (~1.3 GB) are NOT installed here. They download on first use
# into the per-user cache managed by core/models/manager.py, behind a progress
# screen the user opts into.

-r requirements.txt

torch>=2.2.0
transformers>=4.41.0
huggingface_hub>=0.23.0

# 2.0.2 is the first release carrying TimesFM_2p5_200M_torch, the class
# core/models/timesfm.py instantiates.
timesfm[torch]>=2.0.2

# Chronos2Pipeline only exists from 2.2 onwards. The 1.5.x line ships
# ChronosPipeline/ChronosBoltPipeline instead, so an older pin would install
# cleanly and then fail at import time.
chronos-forecasting>=2.2.0
